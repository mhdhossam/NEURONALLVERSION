    "import heapq\n",
    "from typing import Set\n",
    "\n",
    "\n",
    "class NeuronSearchEngine:\n",
    "\n",
    "    def __init__(self, platforms, regions, neurons):\n",
    "        self.platforms = platforms\n",
    "        self.regions = regions\n",
    "        self.neurons = neurons\n",
    "    def resolve_entry(self, platform_uuid: str) -> Set[str]:\n",
    "\n",
    "        platform = self.platforms.get(platform_uuid)\n",
    "\n",
    "        if not platform:\n",
    "            return set()\n",
    "\n",
    "        return platform.region_refs\n",
    "    def expand_regions(self, region_ids: Set[str]) -> Set[str]:\n",
    "\n",
    "        neurons = set()\n",
    "\n",
    "        for rid in region_ids:\n",
    "\n",
    "            region = self.regions.get(rid)\n",
    "            if not region:\n",
    "                continue\n",
    "\n",
    "            # activate region during search\n",
    "            region.activate(0.5)\n",
    "\n",
    "            neurons.update(region.neurons)\n",
    "\n",
    "        return neurons\n",
    "        \n",
    "    def traverse(self, neuron_ids: Set[str], depth=2):\n",
    "\n",
    "        visited = set(neuron_ids)\n",
    "        frontier = set(neuron_ids)\n",
    "\n",
    "        for _ in range(depth):\n",
    "\n",
    "            next_frontier = set()\n",
    "\n",
    "            for nid in frontier:\n",
    "\n",
    "                neuron = self.neurons.get(nid)\n",
    "                if not neuron:\n",
    "                    continue\n",
    "\n",
    "                for syn in neuron.synapses.values():\n",
    "\n",
    "                    target = syn.target   # FIXED\n",
    "\n",
    "                    if target not in visited:\n",
    "                        visited.add(target)\n",
    "                        next_frontier.add(target)\n",
    "\n",
    "            frontier = next_frontier\n",
    "\n",
    "        return visited\n",
    "    def score(self, neuron):\n",
    "\n",
    "        soma = neuron.soma\n",
    "\n",
    "        return (\n",
    "            soma.confidence +\n",
    "            (soma.support * 0.01) -\n",
    "            (soma.mistakes * 0.05) +\n",
    "            (len(neuron.synapses) * 0.01)\n",
    "        )\n",
    "    def search(self, platform_uuid: str, top_k=5):\n",
    "\n",
    "        region_ids = self.resolve_entry(platform_uuid)\n",
    "\n",
    "        neuron_pool = self.expand_regions(region_ids)\n",
    "\n",
    "        expanded = self.traverse(neuron_pool, depth=2)\n",
    "\n",
    "        scored = []\n",
    "\n",
    "        for nid in expanded:\n",
    "\n",
    "            neuron = self.neurons.get(nid)\n",
    "            if not neuron:\n",
    "                continue\n",
    "\n",
    "            score = self.score(neuron)\n",
    "\n",
    "            heapq.heappush(scored, (-score, nid))\n",
    "\n",
    "        results = []\n",
    "\n",
    "        for _ in range(min(top_k, len(scored))):\n",
    "            results.append(heapq.heappop(scored)[1])\n",
    "\n",
    "        return results"

    "===================\n",
    "UUID-based knowledge reconstruction layer for the face recognition pipeline.\n",
    "\n",
    "Architecture:\n",
    "    Query Embedding\n",
    "        ↓  cosine similarity search across all neurons in matching regions\n",
    "    Best Neuron\n",
    "        ↓  neuron.recall('uuid')\n",

    "        ↓  KnowledgeStore.get(uuid)\n",

    "\n",
    "Components:\n",
    "    Neuron          — leaf memory unit; stores arbitrary key/value nucleus data\n",
    "    Soma            — aggregates support signals; drives confidence\n",
    "    Region          — named collection of neurons with a schema; supports query/search\n",
    "    RegionSchema    — domain + roles metadata for a region\n",
    "    Platform        — multi-region container; routes searches by domain\n",
    "    KnowledgeObject — rich identity node (name, metadata, relationships, history)\n",

    "class Support:\n",
    "    \"\"\"\n",
    "    Accumulates reinforcement signals for a neuron.\n",
    "    support  — raw integer count of positive matches\n",
    "    decay    — multiplicative decay applied on negative evidence\n",
    "    \"\"\"\n",
    "\n",
    "    def __init__(self, decay: float = 0.95):\n",
    "        self.support: int = 0\n",
    "        self.decay_factor: float = decay\n",
    "        self._total_queries: int = 0\n",
    "\n",
    "    def reinforce(self, amount: int = 1) -> None:\n",
    "        self.support += amount\n",
    "        self._total_queries += 1\n",
    "\n",
    "    def decay(self) -> None:\n",
    "        \"\"\"Call when neuron was NOT the best match — soft penalty.\"\"\"\n",
    "        self.support = int(self.support * self.decay_factor)\n",
    "        self._total_queries += 1\n",
    "\n",
    "    @property\n",
    "    def reliability(self) -> float:\n",
    "        \"\"\"Fraction of queries where this neuron was reinforced.\"\"\"\n",
    "        if self._total_queries == 0:\n",
    "            return 0.0\n",
    "        return self.support / self._total_queries\n",

    "\n",

    "    \"\"\"\n",
    "    Integrative layer of a Neuron.\n",
    "    Owns the Support accumulator and derives confidence.\n",

    "\n",
    "    def __init__(self):\n",
    "        self.support = Support()\n",
    "        self._base_confidence: float = 0.0\n",
    "\n",

    "        Called when this neuron wins a search with `score`.\n",
    "        Reinforces support and updates confidence.\n",

    "\n",
    "    def __init__(self, schema: RegionSchema):\n",
    "        self.schema: RegionSchema = schema\n",
    "        self._neurons: List[Neuron] = []\n",
    "\n",
    "        # ── Fast-path cache ───────────────────────────────────────────────\n",
    "        # Stacked, L2-normalised embedding matrix: shape (N, DIM), float32.\n",
    "        # Rebuilt lazily on the next search() call after any add().\n",
    "        # A single BLAS sgemv (matrix @ vector) replaces N Python calls.\n",
    "        self._emb_matrix: Optional[np.ndarray] = None   # (N, DIM) normalised\n",
    "        self._matrix_dirty: bool = True                  # invalidated on add()\n",
    "\n",
    "    # ------------------------------------------------------------------\n",
    "    # Internal cache management\n",
    "    # ------------------------------------------------------------------\n",
    "\n",
    "    def _rebuild_matrix(self) -> None:\n",
    "        \"\"\"\n",
    "        Stack all neuron embeddings into a contiguous (N, DIM) float32 matrix\n",
    "        and L2-normalise each row in-place.\n",
    "        Neurons without an 'embedding' key get a zero row (score → 0).\n",
    "        \"\"\"\n",
    "        n = len(self._neurons)\n",
    "        if n == 0:\n",
    "            self._emb_matrix = None\n",
    "            self._matrix_dirty = False\n",
    "            return\n",
    "\n",
    "        dim = None\n",
    "        rows = []\n",
    "        for neuron in self._neurons:\n",
    "            emb = neuron.recall('embedding')\n",
    "            if emb is not None:\n",
    "                e = np.asarray(emb, dtype=np.float32)\n",
    "                if dim is None:\n",
    "                    dim = e.shape[0]\n",
    "                rows.append(e)\n",
    "            else:\n",
    "                rows.append(None)          # placeholder; filled after dim known\n",
    "\n",
    "        if dim is None:\n",
    "            self._emb_matrix = None\n",
    "            self._matrix_dirty = False\n",
    "            return\n",
    "\n",
    "        mat = np.zeros((n, dim), dtype=np.float32)\n",
    "        for i, row in enumerate(rows):\n",
    "            if row is not None:\n",
    "                mat[i] = row\n",
    "\n",
    "        # L2-normalise rows; zero rows stay zero (similarity → 0)\n",
    "        norms = np.linalg.norm(mat, axis=1, keepdims=True)\n",
    "        norms[norms == 0] = 1.0            # avoid divide-by-zero\n",
    "        mat /= norms\n",
    "\n",
    "        self._emb_matrix = mat\n",
    "        self._matrix_dirty = False\n",
    "\n",
    "    def add(self, neuron: Neuron) -> None:\n",
    "        self._neurons.append(neuron)\n",
    "        self._matrix_dirty = True          # invalidate cache\n",
    "\n",
    "    def query(self, **filters) -> List[Neuron]:\n",
    "        \"\"\"\n",
    "        Filter neurons by nucleus key equality.\n",
    "        e.g. region.query(label=\"Mohamed\")\n",
    "        \"\"\"\n",
    "        results = []\n",
    "        for n in self._neurons:\n",
    "            if all(n.recall(k) == v for k, v in filters.items()):\n",

    "        return results\n",
    "\n",
    "    def search(self, embedding: np.ndarray) -> Optional[Dict[str, Any]]:\n",
    "        \"\"\"\n",

    "\n",
    "        Fast path (BLAS):\n",
    "            1. Rebuild normalised matrix if dirty  — O(N·D), amortised\n",
    "            2. L2-normalise query                  — O(D)\n",
    "            3. matrix @ query  (sgemv)             — O(N·D), contiguous BLAS\n",
    "            4. argmax                              — O(N)\n",
    "            5. Fire winner soma, inhibit losers    — O(N) Python, winner only hot\n",
    "\n",
    "        Fires the winner's soma; inhibits all losers (Hebbian competition).\n",
    "\n",
    "        Returns:\n",
    "            {\n",
    "                'neuron':    Neuron,\n",
    "                'score':     float,\n",
    "                'neuron_id': str,\n",
    "            }\n",
    "            or None if region is empty.\n",
    "        \"\"\"\n",
    "        if not self._neurons:\n",
    "            return None\n",
    "\n",
    "        if self._matrix_dirty:\n",
    "            self._rebuild_matrix()\n",
    "\n",
    "        if self._emb_matrix is None:\n",
    "            return None\n",
    "\n",
    "        # ── BLAS path ────────────────────────────────────────────────────\n",
    "        q = np.asarray(embedding, dtype=np.float32)\n",
    "        q_norm = np.linalg.norm(q)\n",
    "        if q_norm == 0.0:\n",
    "            q_norm = 1.0\n",
    "        q = q / q_norm                          # normalised query, shape (D,)\n",
    "\n",
    "        scores = self._emb_matrix @ q           # (N,) — one BLAS sgemv call\n",
    "\n",
    "        best_idx   = int(np.argmax(scores))\n",
    "        best_score = float(scores[best_idx])\n",
    "        best_neuron = self._neurons[best_idx]\n",
    "\n",
    "        # ── Hebbian: winner fires, losers inhibit ─────────────────────────\n",
    "        best_neuron.soma.fire(best_score)\n",
    "        for i, n in enumerate(self._neurons):\n",
    "            if i != best_idx:\n",
    "                n.soma.inhibit()\n",
    "\n",
    "        return {\n",
    "            'neuron':    best_neuron,\n",
    "            'score':     best_score,\n",
    "            'neuron_id': best_neuron.neuron_id,\n",
    "        }\n",
    "\n",
    "    @property\n",

    "\n",
